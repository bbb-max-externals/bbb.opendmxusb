cmake_minimum_required(VERSION 3.19)
if(APPLE AND NOT DEFINED CMAKE_OSX_ARCHITECTURES)
    set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
endif()
if(APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")
endif()
project(bbb.opendmxusb VERSION 0.1.0 LANGUAGES CXX)
if(NOT APPLE)
    message(FATAL_ERROR "The current serial backend requires macOS.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MAX_SDK_BASE "${CMAKE_CURRENT_SOURCE_DIR}/deps/max-sdk-base" CACHE PATH "Existing Cycling '74 max-sdk-base checkout")
if(NOT EXISTS "${MAX_SDK_BASE}/c74support/max-includes/ext.h")
    message(FATAL_ERROR "Initialize the SDK with git submodule update --init, or set MAX_SDK_BASE to an existing checkout")
endif()
add_library(bbb.opendmxusb MODULE source/bbb.opendmxusb.cpp source/serial_mac.cpp)
target_include_directories(bbb.opendmxusb PRIVATE "${MAX_SDK_BASE}/c74support/max-includes")
target_compile_options(bbb.opendmxusb PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-cast-function-type-mismatch)
target_link_options(bbb.opendmxusb PRIVATE "-undefined" "dynamic_lookup")
set_target_properties(bbb.opendmxusb PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION mxo
    PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/externals"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/source/Info.plist.in")
add_custom_command(TARGET bbb.opendmxusb POST_BUILD
    COMMAND codesign --force --sign - "$<TARGET_BUNDLE_DIR:bbb.opendmxusb>")
option(OPEN_DMX_BUILD_TESTS "Build focused frame/lifecycle tests" OFF)
if(OPEN_DMX_BUILD_TESTS)
    enable_testing()
    add_executable(opendmx_tests tests/engine_test.cpp)
    target_include_directories(opendmx_tests PRIVATE source)
    add_test(NAME opendmx_engine COMMAND opendmx_tests)
endif()
